  function extractTotals(lines,text){ let total=null,tax=null,withTax=null,words='';
    for(let i=0;i<lines.length;i++){
      const l=lines[i];
      if(/价税合计/.test(l)){
        let v=nums2(l).filter(n=>n>=0);
        if(!v.length)v=nums2((lines[i+1]||'')+' '+(lines[i+2]||'')).filter(n=>n>=0);
        if(v.length)withTax=v[v.length-1];

        const wm=scope.match(/[零壹贰叁肆伍陆柒捌玖拾佰仟万亿圆元角分整]+/);if(wm)words=wm[0];
      }
      if(/合\s*计/.test(l)&&!/价税/.test(l)){
        let v=nums2(l);
        if(v.length<2&&i+1<lines.length&&!/价税合计/.test(lines[i+1]))v=nums2(l+' '+lines[i+1]);
        if(v.length>=2){total=v[v.length-2];tax=v[v.length-1];}
      }
    }
    const explicitWithTax=text.match(/价税合计[\s\S]{0,80}?[¥￥]?\s*([\d,]+\.\d{1,2})/);if(withTax===null&&explicitWithTax)withTax=Number(explicitWithTax[1].replace(/,/g,''));
    const yuan=unique((text.match(/[¥￥]\s*(-?[\d,]+(?:\.\d{1,2})?)/g)||[]).map(x=>Number(x.replace(/[¥￥\s,]/g,'')))).filter(Number.isFinite);
    const decimals=nums2(text).filter(n=>n>=0&&n<100000000);const pool=[...yuan,...decimals];
    if(withTax===null&&pool.length)withTax=Math.max(...pool);
    if((total===null||tax===null)&&withTax!==null){const smaller=pool.filter(n=>n<withTax-.001).sort((a,b)=>b-a);if(total===null)total=smaller[0]??null;if(tax===null&&total!==null)tax=Number((withTax-total).toFixed(2));}
    if(total===null&&withTax!==null&&tax!==null)total=Number((withTax-tax).toFixed(2));if(tax===null&&withTax!==null&&total!==null)tax=Number((withTax-total).toFixed(2));if(withTax===null&&total!==null&&tax!==null)withTax=Number((total+tax).toFixed(2));
    return{totalAmount:total,totalTax:tax,totalAmountWithTax:withTax,totalAmountInWords:words}; }

  function reconcileTotals(totals,items){
    const amountItems=(items||[]).filter(x=>x.amount!==null&&x.amount!==undefined&&Number.isFinite(Number(x.amount)));
    const taxItems=(items||[]).filter(x=>x.taxAmount!==null&&x.taxAmount!==undefined&&Number.isFinite(Number(x.taxAmount)));
    if(amountItems.length){totals.totalAmount=Number(amountItems.reduce((s,x)=>s+Number(x.amount),0).toFixed(2));}
    if(taxItems.length){totals.totalTax=Number(taxItems.reduce((s,x)=>s+Number(x.taxAmount),0).toFixed(2));}
    if(totals.totalAmount!==null&&totals.totalTax!==null){
      const calculated=Number((Number(totals.totalAmount)+Number(totals.totalTax)).toFixed(2));
      if(totals.totalAmountWithTax===null||Math.abs(Number(totals.totalAmountWithTax)-calculated)>.02)totals.totalAmountWithTax=calculated;
    }
    return totals;
  }
