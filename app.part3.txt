  function determine(r){ if(r.recognitionMethod==='failed'){r.status='failed';r.exceptionNote=r.exceptionNote||'识别失败';return r;} const notes=[]; const missing=[];if(!r.invoiceNumber)missing.push('发票号码');if(!r.invoiceDate)missing.push('开票日期');if(!r.buyerName)missing.push('购买方');if(!r.sellerName)missing.push('销售方');if(r.totalAmountWithTax===null||r.totalAmountWithTax===undefined)missing.push('价税合计');if(missing.length)notes.push('缺少'+missing.join('、')); if(r.totalAmount!==null&&r.totalTax!==null&&r.totalAmountWithTax!==null){const d=Number((Number(r.totalAmount)+Number(r.totalTax)-Number(r.totalAmountWithTax)).toFixed(2));r.amountDiff=d;if(Math.abs(d)>.02)notes.unshift(`金额校验异常（差额${d>0?'+':''}${d.toFixed(2)}元）`);} if(r.amountDiff!==undefined&&Math.abs(r.amountDiff)>.02)r.status='amount_error';else if(r.isDuplicate)r.status='duplicate';else if(missing.length||r.recognitionMethod.startsWith('ocr_'))r.status='pending_review';else r.status='success';r.exceptionNote=notes.join('；')||(r.recognitionMethod.startsWith('ocr_')?'OCR结果建议人工复核':'');return r; }
  function detectDuplicates(){ const map=new Map();state.records.forEach(r=>{r.isDuplicate=false;if(!r.invoiceNumber)return;const k=[r.invoiceCode,r.invoiceNumber,r.invoiceDate,r.sellerName,r.totalAmountWithTax].join('|');if(!map.has(k))map.set(k,[]);map.get(k).push(r);});for(const group of map.values())if(group.length>1)group.forEach(r=>r.isDuplicate=true);state.records.forEach(determine); }

  function groupRows(items,width,height){ const sorted=items.map(i=>({...i,y:height-i.y})).sort((a,b)=>a.y-b.y||a.x-b.x);const rows=[];for(const it of sorted){let row=rows.find(r=>Math.abs(r.y-it.y)<5);if(!row){row={y:it.y,width,items:[]};rows.push(row);}row.items.push(it);}rows.forEach(r=>r.items.sort((a,b)=>a.x-b.x));return rows; }
  async function processPdf(file,fileState,fileIndex,totalFiles){ const pdfjs=await ensurePdf(); const data=await file.arrayBuffer();const pdf=await pdfjs.getDocument({data}).promise;const results=[];for(let p=1;p<=pdf.numPages;p++){ const base=((fileIndex+(p-1)/pdf.numPages)/totalFiles)*100;setProgress(base,'正在识别发票',`${file.name} · 第 ${p}/${pdf.numPages} 页`);const page=await pdf.getPage(p);const viewport=page.getViewport({scale:1});const tc=await page.getTextContent();const items=tc.items.filter(x=>x.str&&x.str.trim()).map(x=>({str:x.str,x:x.transform[4],y:x.transform[5],width:x.width||0}));const rows=groupRows(items,viewport.width,viewport.height);let text=rows.map(r=>r.items.map(i=>i.str).join(' ')).join('\n');let method='pdf_text';if(text.replace(/\s/g,'').length<45){method='ocr_pdf';const canvas=document.createElement('canvas');const vp=page.getViewport({scale:2.35});canvas.width=Math.ceil(vp.width);canvas.height=Math.ceil(vp.height);const ctx=canvas.getContext('2d',{willReadFrequently:true});ctx.fillStyle='#fff';ctx.fillRect(0,0,canvas.width,canvas.height);await page.render({canvasContext:ctx,viewport:vp}).promise;const worker=await createOcrWorker((prog,status)=>setProgress(base+prog*(100/totalFiles/pdf.numPages)*.9,'正在进行 OCR',`${file.name} · ${status}`));const o=await worker.recognize(canvas);text=(o.data.text||'').trim();if(text.replace(/\s/g,'').length<20)throw new Error('OCR未识别到足够文字，请使用更清晰的发票');}
        results.push(parseInvoice(text,{name:file.name,page:p,method,type:'PDF',size:file.size},method==='pdf_text'?rows:null)); }
      return results; }
  async function imageToCanvas(file){const url=URL.createObjectURL(file);try{const img=await new Promise((res,rej)=>{const i=new Image();i.onload=()=>res(i);i.onerror=()=>rej(new Error('图片无法打开'));i.src=url;});const scale=Math.min(1,2800/Math.max(img.naturalWidth,img.naturalHeight));const c=document.createElement('canvas');c.width=Math.max(1,Math.round(img.naturalWidth*scale));c.height=Math.max(1,Math.round(img.naturalHeight*scale));const x=c.getContext('2d',{willReadFrequently:true});x.fillStyle='#fff';x.fillRect(0,0,c.width,c.height);x.drawImage(img,0,0,c.width,c.height);return c;}finally{URL.revokeObjectURL(url);}}
  async function processImage(file,fileIndex,totalFiles){const canvas=await imageToCanvas(file);const worker=await createOcrWorker((prog,status)=>setProgress(((fileIndex+prog*.9)/totalFiles)*100,'正在进行 OCR',`${file.name} · ${status}`));const o=await worker.recognize(canvas);const text=(o.data.text||'').trim();if(text.replace(/\s/g,'').length<20)throw new Error('OCR未识别到足够文字，请使用更清晰的图片');return[parseInvoice(text,{name:file.name,page:1,method:'ocr_image',type:'图片',size:file.size},null)];}
